typedef enum {
    Idle = 0x00u,
    Play_Notice = 0x01,  // 注意音，程序没错但是最好不要再这么做（示例：抵达软件限位但是仍想要移动
    Play_Warning = 0x02, // 警告音，出现问题但是不影响功能正常运行（示例：图传断连但是DR16仍能正常控制
    Play_Error = 0x03,   // 错误音，出现问题导致功能不能运行，但其他功能正常运行（示例：电机断电
    Play_Fatal = 0x04,   // 致命错误音
    Play_WakeUp = 0x05,  // 启动音
    Play_Sleep  = 0x06,  // 关闭音
                         // 7,8保留
    Play_Song1 = 0x09,   // 播放音乐1
    // add
} Music_PlayState_e;

typedef struct{
    Music_PlayState_e play_state;
    uint16_t bpm;
    uint8_t times;
    uint8_t delay_beat;
    const float* song;
